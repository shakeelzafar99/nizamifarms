-- HOW
--   MariaDB has no partial index, so we add a PERSISTENT generated column that equals

--   non-invoice / reversed / rejected rows never collide — only two LIVE invoices for
--   the same order do. The column auto-updates when approval_status changes, so a
--   reversed invoice frees the slot and a re-invoice (reverse-then-post) works fine.
--
-- SAFE: verified 0 orders currently have >1 live invoice (the index would fail to
--   build otherwise), and the only re-invoice path (payment-method change) reverses
--   the old invoice BEFORE posting the new one.
-- =====================================================================

FROM t_fin_ledger
WHERE transaction_type = 'invoice'
  AND order_id IS NOT NULL
  AND approval_status NOT IN ('reversed','rejected')
GROUP BY order_id
HAVING COUNT(*) > 1;

-- Leave ledger: every ADJUSTMENT to a rider's leave balance (on top of the
-- yearly quota). One row = one +/- change, with who/why.
--   source = 'manual'       manager granted extra leaves (+days)
--   source = 'overtime'     earned from overtime hours (+days)          [Phase F]
--   source = 'late_penalty' cumulative lateness deduction (-days)        [Phase F]
--   source = 'half_day'     manager gave a half-day for a late day (-0.5)[Phase F]
-- Built SIGNED (days can be negative) with the full source list from day one
-- so Phase F needs no schema change — it just writes more rows here.
--
-- MySQL note: `IF NOT EXISTS` on CREATE TABLE is standard. The ENUM already
-- lists every source; no ALTER needed later.
CREATE TABLE IF NOT EXISTS t_hr_leave_grant (
  id         INT AUTO_INCREMENT PRIMARY KEY,
  user_id    INT NOT NULL,
  days       DECIMAL(5,1) NOT NULL,                       -- signed: +grant / -penalty
  reason     VARCHAR(200) NULL,
  source     ENUM('manual','overtime','late_penalty','half_day') NOT NULL DEFAULT 'manual',
  effective_date DATE NULL,                               -- which cycle it belongs to; NULL = created_at date
  created_by INT NULL,
  created_at DATETIME NULL,
  updated_at DATETIME NULL,
  INDEX idx_leave_grant_user (user_id),
  INDEX idx_leave_grant_eff (effective_date)
);

-- "Not needed" day tags: a manager marks a rider as not required on a given day.

-- One row per (rider, day); UNIQUE prevents duplicates.
CREATE TABLE IF NOT EXISTS t_ops_day_tag (
  id         INT AUTO_INCREMENT PRIMARY KEY,
  user_id    INT NOT NULL,
  tag_date   DATE NOT NULL,
  tag        VARCHAR(30) NOT NULL DEFAULT 'not_needed',
  note       VARCHAR(200) NULL,
  created_by INT NULL,
  created_at DATETIME NULL,

-- Migration: rider cash-held confirmation at check-out (Jul 2026)
--
-- Adds two columns to the attendance row so a rider can CONFIRM, at check-out,
-- the exact cash he is still holding against him (the "open" Daily Closing

--
--   cash_confirmed_amount = the figure he confirmed
--   cash_confirmed_at     = when he confirmed it
--

-- signal). Idempotent columns; safe to re-run. Run on LOCAL (dev) and PROD.
-- =============================================================================

ALTER TABLE `t_ops_attendance`
    ADD COLUMN IF NOT EXISTS `cash_confirmed_amount` DECIMAL(12,2) NULL DEFAULT NULL
        COMMENT 'Cash the rider confirmed holding at check-out (Daily Closing open amount)',
    ADD COLUMN IF NOT EXISTS `cash_confirmed_at` DATETIME NULL DEFAULT NULL
        COMMENT 'When the rider confirmed the cash-held amount',
    ADD COLUMN IF NOT EXISTS `cash_confirm_status` VARCHAR(20) NULL DEFAULT NULL
        COMMENT "'confirmed' = rider agreed; 'issue' = rider flagged the amount as wrong";
